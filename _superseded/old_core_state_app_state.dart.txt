/// True when some network interface is available. This does not prove the
/// internet is reachable, and nothing in the prototype needs it.
final connectivityProvider = StreamProvider<bool>((ref) async* {
  final c = Connectivity();
  bool online(List<ConnectivityResult> r) => r.any((e) => e != ConnectivityResult.none);
  try {
    yield online(await c.checkConnectivity());
    yield* c.onConnectivityChanged.map(online);
  } catch (_) {
    yield true;
  }
});

class EligibleReview {
  final String contextId;
  final String label;
  final String workerId;
  const EligibleReview(this.contextId, this.label, this.workerId);
}

class AppData {
  final bool onboarded;
  final UserRole? chosenRole;
  final AppUser? user;
  final List<AppUser> accounts;
  final String demoWorkerId;
  final List<WorkerProfile> workers;
  final List<Job> jobs;
  final List<JobApplication> applications;
  final List<Booking> bookings;
  final List<Review> reviews;
  final List<Conversation> conversations;
  final List<NotificationItem> notifications;
  final List<QuizAttempt> attempts;
  final List<String> favourites;
  final List<String> recentWorkers;
  final List<String> recentTrades;
  final UserSettings settings;
  final List<String> pending;
  final WorkerProfile? draft;

  const AppData({
    this.onboarded = false,
    this.chosenRole,
    this.user,
    this.accounts = const [],
    this.demoWorkerId = kDefaultDemoWorkerId,
    this.workers = const [],
    this.jobs = const [],
    this.applications = const [],
    this.bookings = const [],
    this.reviews = const [],
    this.conversations = const [],
    this.notifications = const [],
    this.attempts = const [],
    this.favourites = const [],
    this.recentWorkers = const [],
    this.recentTrades = const [],
    this.settings = const UserSettings(),
    this.pending = const [],
    this.draft,
  });

  AppData copyWith({
    bool? onboarded,
    UserRole? chosenRole,
    AppUser? user,
    bool clearUser = false,
    List<AppUser>? accounts,
    String? demoWorkerId,
    List<WorkerProfile>? workers,
    List<Job>? jobs,
    List<JobApplication>? applications,
    List<Booking>? bookings,
    List<Review>? reviews,
    List<Conversation>? conversations,
    List<NotificationItem>? notifications,
    List<QuizAttempt>? attempts,
    List<String>? favourites,
    List<String>? recentWorkers,
    List<String>? recentTrades,
    UserSettings? settings,
    List<String>? pending,
    WorkerProfile? draft,
    bool clearDraft = false,
  }) =>
      AppData(
        onboarded: onboarded ?? this.onboarded,
        chosenRole: chosenRole ?? this.chosenRole,
        user: clearUser ? null : (user ?? this.user),
        accounts: accounts ?? this.accounts,
        demoWorkerId: demoWorkerId ?? this.demoWorkerId,
        workers: workers ?? this.workers,
        jobs: jobs ?? this.jobs,
        applications: applications ?? this.applications,
        bookings: bookings ?? this.bookings,
        reviews: reviews ?? this.reviews,
        conversations: conversations ?? this.conversations,
        notifications: notifications ?? this.notifications,
        attempts: attempts ?? this.attempts,
        favourites: favourites ?? this.favourites,
        recentWorkers: recentWorkers ?? this.recentWorkers,
        recentTrades: recentTrades ?? this.recentTrades,
        settings: settings ?? this.settings,
        pending: pending ?? this.pending,
        draft: clearDraft ? null : (draft ?? this.draft),
      );
}

/// Read-only helpers that derive values from [AppData].
extension AppSelectors on AppData {
  UserRole get role => user?.role ?? chosenRole ?? UserRole.client;
  bool get isWorker => role == UserRole.worker;

  String? get myWorkerId =>
      user == null ? null : (user!.isDemo ? demoWorkerId : kMyWorkerId);

  WorkerProfile? workerById(String? id) {
    if (id == null) return null;
    for (final w in workers) {
      if (w.id == id) return w;
    }
    return null;
  }

  WorkerProfile? get myWorker => workerById(myWorkerId);

  Job? jobById(String id) {
    for (final j in jobs) {
      if (j.id == id) return j;
    }
    return null;
  }

  Booking? bookingById(String id) {
    for (final b in bookings) {
      if (b.id == id) return b;
    }
    return null;
  }

  Conversation? conversationById(String id) {
    for (final c in conversations) {
      if (c.id == id) return c;
    }
    return null;
  }

  List<Review> reviewsFor(String workerId) =>
      reviews.where((r) => r.workerId == workerId).toList()
        ..sort((a, b) => b.createdAt.compareTo(a.createdAt));

  RatingSummary ratingFor(WorkerProfile w) => summarizeRatings(w, reviews);

  Set<String> badgesFor(WorkerProfile w) =>
      computeBadges(worker: w, reviews: reviews, attempts: attempts);

  List<NotificationItem> get visibleNotifications {
    final l = notifications.where((n) {
      if (n.forRole != role) return false;
      switch (n.type.category) {
        case 'messages':
          return settings.notifyMessages;
        case 'reminders':
          return settings.notifyReminders;
        default:
          return settings.notifyJobs;
      }
    }).toList()
      ..sort((a, b) => b.at.compareTo(a.at));
    return l;
  }

  int get unreadNotifications => visibleNotifications.where((n) => !n.read).length;

  List<Conversation> get myConversations {
    final l = conversations.where((c) {
      if (isWorker) return c.workerId == myWorkerId;
      return user != null && c.clientId == user!.id;
    }).toList()
      ..sort((a, b) => (b.last?.at ?? DateTime(2000)).compareTo(a.last?.at ?? DateTime(2000)));
    return l;
  }

  int unreadFor(Conversation c) => isWorker ? c.unreadForWorker : c.unreadForClient;
  int get unreadMessages => myConversations.fold(0, (a, c) => a + unreadFor(c));

  List<JobApplication> applicantsFor(String jobId) =>
      applications.where((a) => a.jobId == jobId).toList()
        ..sort((a, b) => b.appliedAt.compareTo(a.appliedAt));

  JobApplication? applicationFor(String jobId, String? workerId) {
    for (final a in applications) {
      if (a.jobId == jobId && a.workerId == workerId) return a;
    }
    return null;
  }

  bool hasReviewed(String contextId) =>
      user != null && reviews.any((r) => r.contextId == contextId && r.clientId == user!.id);

  /// Completed jobs and bookings that this client can still review.
  List<EligibleReview> eligibleReviews({String? workerId}) {
    final u = user;
    if (u == null) return const [];
    final out = <EligibleReview>[];
    for (final j in jobs) {
      if (j.clientId != u.id || j.status != JobStatus.completed || hasReviewed(j.id)) continue;
      for (final a in applications) {
        if (a.jobId == j.id && a.status == ApplicationStatus.accepted) {
          if (workerId == null || a.workerId == workerId) {
            out.add(EligibleReview(j.id, j.title, a.workerId));
          }
          break;
        }
      }
    }
    for (final b in bookings) {
      if (b.clientId != u.id || b.status != BookingStatus.completed || hasReviewed(b.id)) continue;
      if (workerId == null || b.workerId == workerId) {
        out.add(EligibleReview(b.id, b.service, b.workerId));
      }
    }
    return out;
  }

  bool isFavourite(String id) => favourites.contains(id);
}

final appProvider = NotifierProvider<AppController, AppData>(AppController.new);

class AppController extends Notifier<AppData> {
  LocalStore get _store => ref.read(storeProvider);

  static const _kSeeded = 'sb.seeded.v1';

  @override
  AppData build() => _load();

  // ----- loading and persistence ------------------------------------------

  void _seed() {
    final s = DemoSeed.build(DateTime.now());
    _store.writeList('sb.workers', s.workers.map((e) => e.toJson()).toList());
    _store.writeList('sb.jobs', s.jobs.map((e) => e.toJson()).toList());
    _store.writeList('sb.apps', s.applications.map((e) => e.toJson()).toList());
    _store.writeList('sb.bookings', s.bookings.map((e) => e.toJson()).toList());
    _store.writeList('sb.reviews', s.reviews.map((e) => e.toJson()).toList());
    _store.writeList('sb.convs', s.conversations.map((e) => e.toJson()).toList());
    _store.writeList('sb.notifs', s.notifications.map((e) => e.toJson()).toList());
    _store.setBool(_kSeeded, true);
  }

  AppData _load() {
    final s = _store;
    // Seed once. The flag prevents duplicate demo records on every launch.
    if (!s.getBool(_kSeeded)) _seed();
    final userMap = s.readMap('sb.user');
    AppUser? user;
    if (userMap != null) {
      try {
        user = AppUser.fromJson(userMap);
      } catch (_) {}
    }
    WorkerProfile? draft;
    final draftMap = s.readMap('sb.draft');
    if (draftMap != null) {
      try {
        draft = WorkerProfile.fromJson(draftMap);
      } catch (_) {}
    }
    return AppData(
      onboarded: s.getBool('sb.onboarded'),
      chosenRole: enumFrom<UserRole>(UserRole.values, s.getString('sb.chosenRole'), UserRole.client),
      user: user,
      accounts: s.readList('sb.accounts', AppUser.fromJson),
      demoWorkerId: s.getString('sb.demoWorker') ?? kDefaultDemoWorkerId,
      workers: s.readList('sb.workers', WorkerProfile.fromJson),
      jobs: s.readList('sb.jobs', Job.fromJson),
      applications: s.readList('sb.apps', JobApplication.fromJson),
      bookings: s.readList('sb.bookings', Booking.fromJson),
      reviews: s.readList('sb.reviews', Review.fromJson),
      conversations: s.readList('sb.convs', Conversation.fromJson),
      notifications: s.readList('sb.notifs', NotificationItem.fromJson),
      attempts: s.readList('sb.attempts', QuizAttempt.fromJson),
      favourites: s.getStrings('sb.favs'),
      recentWorkers: s.getStrings('sb.recentW'),
      recentTrades: s.getStrings('sb.recentT'),
      settings: UserSettings.fromJson(s.readMap('sb.settings') ?? {}),
      pending: s.getStrings('sb.pending'),
      draft: draft,
    );
  }

  void _persist() {
    final s = _store;
    final d = state;
    s.setBool('sb.onboarded', d.onboarded);
    if (d.chosenRole != null) s.setString('sb.chosenRole', d.chosenRole!.name);
    s.writeMap('sb.user', d.user?.toJson());
    s.writeList('sb.accounts', d.accounts.map((e) => e.toJson()).toList());
    s.setString('sb.demoWorker', d.demoWorkerId);
    s.writeList('sb.workers', d.workers.map((e) => e.toJson()).toList());
    s.writeList('sb.jobs', d.jobs.map((e) => e.toJson()).toList());
    s.writeList('sb.apps', d.applications.map((e) => e.toJson()).toList());
    s.writeList('sb.bookings', d.bookings.map((e) => e.toJson()).toList());
    s.writeList('sb.reviews', d.reviews.map((e) => e.toJson()).toList());
    s.writeList('sb.convs', d.conversations.map((e) => e.toJson()).toList());
    s.writeList('sb.notifs', d.notifications.map((e) => e.toJson()).toList());
    s.writeList('sb.attempts', d.attempts.map((e) => e.toJson()).toList());
    s.setStrings('sb.favs', d.favourites);
    s.setStrings('sb.recentW', d.recentWorkers);
    s.setStrings('sb.recentT', d.recentTrades);
    s.writeMap('sb.settings', d.settings.toJson());
    s.setStrings('sb.pending', d.pending);
    s.writeMap('sb.draft', d.draft?.toJson());
  }

  /// Applies a change, records it as a pending (not yet synced) change and saves.
  void _apply(AppData next, [String? change]) {
    var pend = next.pending;
    if (change != null) {
      pend = [...pend, change];
      if (pend.length > 200) pend = pend.sublist(pend.length - 200);
    }
    state = next.copyWith(pending: pend);
    _persist();
  }

  String _id(String p) => '${p}_${DateTime.now().microsecondsSinceEpoch}';

  // ----- onboarding and account -------------------------------------------

  void completeOnboarding() => _apply(state.copyWith(onboarded: true));

  void setRole(UserRole role) {
    final u = state.user;
    _apply(state.copyWith(chosenRole: role, user: u?.copyWith(role: role)));
  }

  void setDemoWorker(String workerId) => _apply(state.copyWith(demoWorkerId: workerId));

  void signInDemo(UserRole role) {
    final repo = ref.read(authRepoProvider);
    final name = state.workerById(state.demoWorkerId)?.name;
    _apply(state.copyWith(
      onboarded: true,
      chosenRole: role,
      user: repo.demoUser(role, workerName: name),
    ));
  }

  /// Returns an error message, or null on success.
  String? register({
    required String name,
    required String phone,
    String email = '',
    required String password,
  }) {
    final repo = ref.read(authRepoProvider);
    final pwErr = repo.validatePassword(password);
    if (pwErr != null) return pwErr;
    final existing = repo.findAccount(state.accounts, phone) ??
        (email.isNotEmpty ? repo.findAccount(state.accounts, email) : null);
    if (existing != null) return 'An account with these details already exists on this device. Try signing in.';
    final user = repo.createAccount(name: name, phone: phone, email: email, role: state.role);
    _apply(state.copyWith(onboarded: true, user: user, accounts: [...state.accounts, user]), 'account:${user.id}:create');
    return null;
  }

  String? signIn({required String identifier, required String password}) {
    final repo = ref.read(authRepoProvider);
    final pwErr = repo.validatePassword(password);
    if (pwErr != null) return pwErr;
    final acc = repo.findAccount(state.accounts, identifier);
    if (acc == null) {
      return 'No account found on this device. Create an account or continue as demo.';
    }
    _apply(state.copyWith(onboarded: true, user: acc.copyWith(role: state.chosenRole ?? acc.role)));
    return null;
  }

  void signOut() => _apply(state.copyWith(clearUser: true));

  void updateAccount({String? name, String? phone, String? email, String? photoPath}) {
    final u = state.user;
    if (u == null) return;
    final nu = u.copyWith(name: name, phone: phone, email: email, photoPath: photoPath);
    var workers = state.workers;
    if (!u.isDemo && name != null) {
      workers = [
        for (final w in workers) w.id == kMyWorkerId ? w.copyWith(name: name) : w,
      ];
    }
    _apply(
      state.copyWith(
        user: nu,
        workers: workers,
        accounts: [for (final a in state.accounts) a.id == nu.id ? nu : a],
      ),
      'account:${u.id}:update',
    );
  }

  /// Wipes only SkillBridge data (and the app's own media folder), then re-seeds.
  Future<void> clearDemoData() async {
    await _store.clearApp();
    await ref.read(mediaStoreProvider).clearAll();
    _seed();
    state = _load();
  }

  // ----- worker profile ----------------------------------------------------

  void saveDraft(WorkerProfile draft) => _apply(state.copyWith(draft: draft));

  void saveWorkerProfile(WorkerProfile p) {
    final exists = state.workers.any((w) => w.id == p.id);
    final workers = exists
        ? [for (final w in state.workers) w.id == p.id ? p : w]
        : [...state.workers, p];
    _apply(state.copyWith(workers: workers, clearDraft: true), 'worker:${p.id}:update');
  }

  void updateWorker(String id, WorkerProfile Function(WorkerProfile) f) {
    _apply(
      state.copyWith(workers: [for (final w in state.workers) w.id == id ? f(w) : w]),
      'worker:$id:update',
    );
  }

  void upsertPortfolioItem(String workerId, PortfolioItem item) {
    updateWorker(workerId, (w) {
      final has = w.portfolio.any((p) => p.id == item.id);
      return w.copyWith(
        portfolio: has
            ? [for (final p in w.portfolio) p.id == item.id ? item : p]
            : [...w.portfolio, item],
      );
    });
  }

  void removePortfolioItem(String workerId, String itemId) {
    updateWorker(workerId,
        (w) => w.copyWith(portfolio: w.portfolio.where((p) => p.id != itemId).toList()));
  }

  // ----- discovery ---------------------------------------------------------

  void toggleFavourite(String workerId) {
    final fav = [...state.favourites];
    fav.contains(workerId) ? fav.remove(workerId) : fav.add(workerId);
    _apply(state.copyWith(favourites: fav));
  }

  void markViewed(String workerId) {
    final l = [workerId, ...state.recentWorkers.where((e) => e != workerId)].take(10).toList();
    _apply(state.copyWith(recentWorkers: l));
  }

  void markTradeViewed(String tradeId) {
    final l = [tradeId, ...state.recentTrades.where((e) => e != tradeId)].take(6).toList();
    _apply(state.copyWith(recentTrades: l));
  }

  // ----- notifications helper ---------------------------------------------

  NotificationItem _note(UserRole role, NotifType t, String title, String body, {String? route}) =>
      NotificationItem(
        id: _id('n'),
        forRole: role,
        type: t,
        title: title,
        body: body,
        at: DateTime.now(),
        route: route,
      );

  void markNotificationRead(String id) {
    _apply(state.copyWith(notifications: [
      for (final n in state.notifications) n.id == id ? n.copyWith(read: true) : n,
    ]));
  }

  void markAllNotificationsRead() {
    final role = state.role;
    _apply(state.copyWith(notifications: [
      for (final n in state.notifications) n.forRole == role ? n.copyWith(read: true) : n,
    ]));
  }

  void updateSettings(UserSettings s) => _apply(state.copyWith(settings: s));

  // ----- jobs --------------------------------------------------------------

  Job postJob({
    required String title,
    required String tradeId,
    required String description,
    required String town,
    String neighbourhood = '',
    String completeBy = '',
    required JobUrgency urgency,
    double? budgetMin,
    double? budgetMax,
    required PricingType pricingPref,
    List<String> photoPaths = const [],
    required String contactMethod,
  }) {
    final u = state.user!;
    final job = Job(
      id: _id('j'),
      title: title.trim(),
      tradeId: tradeId,
      description: description.trim(),
      province: provinceOf(town),
      town: town,
      neighbourhood: neighbourhood.trim(),
      completeBy: completeBy,
      urgency: urgency,
      budgetMin: budgetMin,
      budgetMax: budgetMax,
      pricingPref: pricingPref,
      photoPaths: photoPaths,
      contactMethod: contactMethod,
      clientId: u.id,
      clientName: u.name,
      postedAt: DateTime.now(),
    );
    final notes = <NotificationItem>[];
    final mine = state.myWorker;
    if (mine != null && mine.tradeId == tradeId) {
      notes.add(_note(UserRole.worker, NotifType.jobMatch, 'New job match',
          '${job.title} in ${job.town} matches your trade.',
          route: '/job/${job.id}'));
    }
    _apply(
      state.copyWith(jobs: [job, ...state.jobs], notifications: [...notes, ...state.notifications]),
      'job:${job.id}:create',
    );
    return job;
  }

  String provinceOf(String town) {
    for (final w in state.workers) {
      if (w.town == town) return w.province;
    }
    return 'Lusaka';
  }

  /// Returns an error message, or null on success.
  String? applyToJob(String jobId, String message) {
    final wid = state.myWorkerId;
    final job = state.jobById(jobId);
    if (wid == null || job == null) return 'This job could not be found.';
    if (job.status != JobStatus.open) return 'This job is no longer open for applications.';
    final existing = state.applicationFor(jobId, wid);
    if (existing != null && existing.status != ApplicationStatus.withdrawn) {
      return 'You have already applied to this job.';
    }
    final app = JobApplication(
      id: existing?.id ?? _id('a'),
      jobId: jobId,
      workerId: wid,
      message: message.trim(),
      appliedAt: DateTime.now(),
    );
    final apps = existing == null
        ? [...state.applications, app]
        : [for (final a in state.applications) a.id == existing.id ? app : a];
    final who = state.myWorker?.name ?? 'A worker';
    _apply(
      state.copyWith(applications: apps, notifications: [
        _note(UserRole.client, NotifType.application, 'New application',
            '$who applied to "${job.title}".',
            route: '/job/$jobId'),
        ...state.notifications,
      ]),
      'application:${app.id}:create',
    );
    return null;
  }

  String? withdrawApplication(String appId) {
    final a = state.applications.where((e) => e.id == appId).firstOrNull;
    if (a == null) return 'Application not found.';
    if (a.status != ApplicationStatus.pending) return 'Only pending applications can be withdrawn.';
    _apply(
      state.copyWith(applications: [
        for (final x in state.applications)
          x.id == appId ? x.copyWith(status: ApplicationStatus.withdrawn) : x,
      ]),
      'application:$appId:withdraw',
    );
    return null;
  }

  String? decideApplication(String appId, {required bool accept}) {
    final a = state.applications.where((e) => e.id == appId).firstOrNull;
    if (a == null) return 'Application not found.';
    final job = state.jobById(a.jobId);
    if (job == null) return 'Job not found.';
    if (a.status != ApplicationStatus.pending) return 'This application has already been decided.';
    if (job.status != JobStatus.open) return 'This job is no longer open.';
    final apps = [
      for (final x in state.applications)
        if (x.id == appId)
          x.copyWith(status: accept ? ApplicationStatus.accepted : ApplicationStatus.declined)
        else if (accept && x.jobId == a.jobId && x.status == ApplicationStatus.pending)
          x.copyWith(status: ApplicationStatus.declined)
        else
          x,
    ];
    final jobs = accept
        ? [for (final j in state.jobs) j.id == job.id ? j.copyWith(status: JobStatus.inProgress) : j]
        : state.jobs;
    final note = _note(
      UserRole.worker,
      NotifType.applicationStatus,
      accept ? 'Application accepted' : 'Application declined',
      accept
          ? 'Your application for "${job.title}" was accepted.'
          : 'Your application for "${job.title}" was not selected.',
      route: '/job/${job.id}',
    );
    _apply(
      state.copyWith(applications: apps, jobs: jobs, notifications: [note, ...state.notifications]),
      'application:$appId:${accept ? 'accept' : 'decline'}',
    );
    return null;
  }

  String? completeJob(String jobId) {
    final job = state.jobById(jobId);
    if (job == null) return 'Job not found.';
    if (job.status != JobStatus.inProgress) return 'Only jobs in progress can be completed.';
    final acc = state.applications
        .where((a) => a.jobId == jobId && a.status == ApplicationStatus.accepted)
        .firstOrNull;
    final workerName = state.workerById(acc?.workerId)?.name ?? 'the worker';
    _apply(
      state.copyWith(
        jobs: [for (final j in state.jobs) j.id == jobId ? j.copyWith(status: JobStatus.completed) : j],
        notifications: [
          if (acc != null)
            _note(UserRole.client, NotifType.reviewReminder, 'Leave a review',
                'Your job "${job.title}" is complete. Review $workerName.',
                route: '/review/$jobId?workerId=${acc.workerId}'),
          ...state.notifications,
        ],
      ),
      'job:$jobId:complete',
    );
    return null;
  }

  String? cancelJob(String jobId) {
    final job = state.jobById(jobId);
    if (job == null) return 'Job not found.';
    if (job.status == JobStatus.completed || job.status == JobStatus.cancelled) {
      return 'This job can no longer be cancelled.';
    }
    _apply(
      state.copyWith(
        jobs: [for (final j in state.jobs) j.id == jobId ? j.copyWith(status: JobStatus.cancelled) : j],
        applications: [
          for (final a in state.applications)
            a.jobId == jobId &&
                    (a.status == ApplicationStatus.pending || a.status == ApplicationStatus.accepted)
                ? a.copyWith(status: ApplicationStatus.declined)
                : a,
        ],
      ),
      'job:$jobId:cancel',
    );
    return null;
  }

  // ----- bookings ----------------------------------------------------------

  Booking createBooking(ServiceRequest r) {
    final u = state.user!;
    final b = Booking(
      id: _id('b'),
      workerId: r.workerId,
      clientId: u.id,
      clientName: u.name,
      service: r.service,
      description: r.description,
      preferredDate: r.preferredDate,
      preferredTime: r.preferredTime,
      location: r.location,
      budget: r.budget,
      notes: r.notes,
      createdAt: DateTime.now(),
    );
    _apply(
      state.copyWith(bookings: [b, ...state.bookings], notifications: [
        _note(UserRole.worker, NotifType.serviceRequest, 'New service request',
            '${u.name} requested: ${r.service}.',
            route: '/booking/${b.id}'),
        ...state.notifications,
      ]),
      'booking:${b.id}:create',
    );
    return b;
  }

  String? setBookingStatus(String id, BookingStatus to) {
    final b = state.bookingById(id);
    if (b == null) return 'Booking not found.';
    final allowed = allowedBookingTransitions(b.status, state.role);
    if (!allowed.contains(to)) {
      return 'A ${b.status.label.toLowerCase()} booking cannot be changed to ${to.label.toLowerCase()}.';
    }
    final toRole = state.isWorker ? UserRole.client : UserRole.worker;
    _apply(
      state.copyWith(
        bookings: [for (final x in state.bookings) x.id == id ? x.copyWith(status: to) : x],
        notifications: [
          _note(toRole, NotifType.booking, 'Booking ${to.label.toLowerCase()}',
              '"${b.service}" is now ${to.label.toLowerCase()}.',
              route: '/booking/$id'),
          if (to == BookingStatus.completed)
            _note(UserRole.client, NotifType.reviewReminder, 'Leave a review',
                'Your booking "${b.service}" is complete. Share a review.',
                route: '/review/$id?workerId=${b.workerId}'),
          ...state.notifications,
        ],
      ),
      'booking:$id:${to.name}',
    );
    return null;
  }

  // ----- messaging ---------------------------------------------------------

  String openConversation({
    required String workerId,
    required String clientId,
    required String clientName,
  }) {
    for (final c in state.conversations) {
      if (c.workerId == workerId && c.clientId == clientId) return c.id;
    }
    final c = Conversation(
      id: _id('cv'),
      workerId: workerId,
      clientId: clientId,
      clientName: clientName,
    );
    _apply(state.copyWith(conversations: [...state.conversations, c]));
    return c.id;
  }

  /// Local demo messaging only. Nothing is sent to another device.
  void sendMessage(String convId, String text) {
    final t = text.trim();
    if (t.isEmpty) return;
    final fromWorker = state.isWorker;
    final convs = <Conversation>[];
    for (final c in state.conversations) {
      if (c.id != convId) {
        convs.add(c);
        continue;
      }
      convs.add(c.copyWith(
        messages: [...c.messages, Message(id: _id('m'), fromWorker: fromWorker, text: t, at: DateTime.now())],
        unreadForWorker: fromWorker ? c.unreadForWorker : c.unreadForWorker + 1,
        unreadForClient: fromWorker ? c.unreadForClient + 1 : c.unreadForClient,
      ));
    }
    _apply(state.copyWith(conversations: convs), 'message:$convId:send');
  }

  void markConversationRead(String convId) {
    final c = state.conversationById(convId);
    if (c == null || state.unreadFor(c) == 0) return;
    final isW = state.isWorker;
    _apply(state.copyWith(conversations: [
      for (final x in state.conversations)
        x.id == convId
            ? x.copyWith(unreadForWorker: isW ? 0 : x.unreadForWorker, unreadForClient: isW ? x.unreadForClient : 0)
            : x,
    ]));
  }

  // ----- reviews -----------------------------------------------------------

  /// Returns an error message, or null on success.
  String? submitReview({
    required String workerId,
    required String contextId,
    required String contextLabel,
    required int rating,
    required String text,
    List<String> tags = const [],
  }) {
    final u = state.user;
    if (u == null) return 'Please sign in first.';
    if (state.hasReviewed(contextId)) return 'You have already reviewed this job.';
    final eligible = state.eligibleReviews(workerId: workerId).any((e) => e.contextId == contextId);
    if (!eligible) return 'You can only review a worker after a completed job or booking.';
    if (rating < 1 || rating > 5) return 'Choose a star rating from 1 to 5.';
    final r = Review(
      id: _id('r'),
      workerId: workerId,
      clientId: u.id,
      clientName: u.name,
      rating: rating,
      text: text.trim(),
      tags: tags,
      createdAt: DateTime.now(),
      contextId: contextId,
      contextLabel: contextLabel,
    );
    _apply(state.copyWith(reviews: [r, ...state.reviews]), 'review:${r.id}:create');
    return null;
  }

  // ----- quizzes -----------------------------------------------------------

  QuizAttempt recordAttempt(String quizId, List<int> answers) {
    final quiz = quizById(quizId);
    final wid = state.myWorkerId ?? kMyWorkerId;
    final worker = state.workerById(wid);
    final before = worker == null ? <String>{} : state.badgesFor(worker);
    final attempt = QuizAttempt(
      id: _id('qa'),
      quizId: quizId,
      workerId: wid,
      score: scoreQuiz(quiz, answers),
      total: quiz.questions.length,
      answers: answers,
      at: DateTime.now(),
    );
    final next = state.copyWith(attempts: [...state.attempts, attempt]);
    final after = worker == null ? <String>{} : next.badgesFor(worker);
    final gained = after.difference(before);
    _apply(
      next.copyWith(notifications: [
        for (final id in gained)
          _note(UserRole.worker, NotifType.quiz, 'Badge earned', 'You earned the ${badgeById(id).name} badge.',
              route: '/badges'),
        ...next.notifications,
      ]),
      'quiz:${attempt.id}:attempt',
    );
    return attempt;
  }
